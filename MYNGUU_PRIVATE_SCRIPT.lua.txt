-- ===================== HELPERS =====================
local function notify(title, content, duration)
    Fluent:Notify({
        Title = title or "MYNGUU",
        Content = content or "",
        Duration = duration or 3
    })
end

local function getChar()
    return LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
end

local function getHRP()
    local c = getChar()
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function getHum()
    local c = getChar()
    return c and c:FindFirstChildOfClass("Humanoid")
end

local function equipTool(name)
    local char = getChar()
    local hum = getHum()
    local bag = LocalPlayer:FindFirstChild("Backpack")
    if not char or not hum then return false end
    if char:FindFirstChild(name) then return true end
    if bag then
        local t = bag:FindFirstChild(name)
        if t then
            hum:EquipTool(t)
            return true
        end
    end
    return false
end

local function fireRep()
    pcall(function()
        muscleEvent:FireServer("rep")
    end)
end

local function firePunch()
    pcall(function()
        muscleEvent:FireServer("punch", "leftHand")
        muscleEvent:FireServer("punch", "rightHand")
        muscleEvent:FireServer("punch")
    end)
    local char = getChar()
    local punch = char and char:FindFirstChild("Punch")
    if punch then
        pcall(function() punch:Activate() end)
    end
end

-- merge live folder
pcall(function()
    local folder = ReplicatedStorage:FindFirstChild("cPetShopFolder")
    local shared = ReplicatedStorage:FindFirstChild("shared")
    local runtime = shared and shared:FindFirstChild("runtime")
    local folder2 = runtime and runtime:FindFirstChild("cPetShopFolder")
    local seen = {}
    for _, n in ipairs(petList) do seen[n] = true end
    for _, f in ipairs({folder, folder2}) do
        if f then
            for _, ch in ipairs(f:GetChildren()) do
                if not seen[ch.Name] then
                    petList[#petList + 1] = ch.Name
                    seen[ch.Name] = true
                end
            end
        end
    end
    table.sort(petList)
end)

local selectedPet = petList[1] or "Apex Overlord"

    Values = petList,
    Multi = false,
    Default = 1,
    Callback = function(v)
        selectedPet = v
    end
})

local function buyPet(name)
    local ok = false
    pcall(function()
        local remote = rEvents:FindFirstChild("cPetShopRemote") or ReplicatedStorage:FindFirstChild("cPetShopRemote")
        local shared = ReplicatedStorage:FindFirstChild("shared")
        local runtime = shared and shared:FindFirstChild("runtime")
        local folder = (runtime and runtime:FindFirstChild("cPetShopFolder"))
            or ReplicatedStorage:FindFirstChild("cPetShopFolder")
        local petObj = folder and folder:FindFirstChild(name)
        if remote and petObj then
            if remote:IsA("RemoteFunction") then
                remote:InvokeServer(petObj)
            else
                remote:FireServer(petObj)
            end
            ok = true
        end
    end)
    return ok
end
